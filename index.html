Koneksi Cloud Database (Supabase PostgreSQL)

Hubungkan ke proyek Supabase gratis Anda agar soal baru, daftar anggota siswa, dan nilai murid di HP langsung tersinkronisasi otomatis.

Skrip SQL Tabel Nilai & Bank Soal 
-- Jalankan di SQL Editor Supabase Anda
CREATE TABLE IF NOT EXISTS hasil_belajar_3d (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  kode_kelas TEXT NOT NULL,
  nama_siswa TEXT NOT NULL,
  no_absen TEXT NOT NULL,
  skor INTEGER NOT NULL,
  akurasi TEXT,
  waktu_selesai TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS bank_soal_3d (
  kode_kelas TEXT PRIMARY KEY,
  daftar_soal JSONB NOT NULL,
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

ALTER TABLE hasil_belajar_3d ENABLE ROW LEVEL SECURITY;
ALTER TABLE bank_soal_3d ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Izinkan semua murid kirim nilai" ON hasil_belajar_3d FOR INSERT WITH CHECK (true);
CREATE POLICY "Izinkan guru baca nilai kelas" ON hasil_belajar_3d FOR SELECT USING (true);
CREATE POLICY "Izinkan baca soal kelas" ON bank_soal_3d FOR SELECT USING (true);
CREATE POLICY "Izinkan simpan soal kelas" ON bank_soal_3d FOR INSERT WITH CHECK (true);
CREATE POLICY "Izinkan update soal kelas" ON bank_soal_3d FOR UPDATE USING (true);
