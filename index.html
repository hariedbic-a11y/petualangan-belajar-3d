Analisis Tingkat Kesukaran Butir Soal

Evaluasi butir soal berdasarkan persentase ketepatan jawaban murid di dalam dunia 3D.

1. Konfigurasi API Supabase

Hubungkan ke proyek Supabase gratis Anda agar soal & nilai murid tersinkronisasi lintas perangkat.

2. Skrip SQL Tabel Supabase

Jalankan sekali di menu SQL Editor Supabase Anda:

CREATE TABLE IF NOT EXISTS hasil_belajar_3d (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  kode_kelas TEXT NOT NULL,
  nama_siswa TEXT NOT NULL,
  no_absen TEXT NOT NULL,
  skor INTEGER NOT NULL,
  akurasi TEXT,
  waktu_selesai TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS bank_soal_3d (
  kode_kelas TEXT PRIMARY KEY,
  daftar_soal JSONB NOT NULL,
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

ALTER TABLE hasil_belajar_3d ENABLE ROW LEVEL SECURITY;
ALTER TABLE bank_soal_3d ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Publik baca tulis nilai" ON hasil_belajar_3d FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Publik baca tulis soal" ON bank_soal_3d FOR ALL USING (true) WITH CHECK (true);
